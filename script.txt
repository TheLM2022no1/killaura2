local b='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+'
local function base64Decode(data)
    data = string.gsub(data, '[^'..b..'=]', '')
    return (datagsub('.', function(x)
        if x == '=' then return '' end
        local r,f='',(bfind(x)-1)
        for i=6,1,-1 do r=r..(f%2^i - f%2^(i-1)  0 and '1' or '0') end
        return r;
    end)gsub('%d%d%d%d%d%d%d%d', function(x)
        if #x ~= 8 then return '' end
        local c = 0
        for i = 1, 8 do
            c = c + (xsub(i,i)=='1' and 2^(8-i) or 0)
        end
        return string.char(c)
    end))
end

-- ฟังก์ชันดาวน์โหลด
local function tryDownload(url)
    return pcall(function()
        return gameHttpGet(url)
    end)
end

-- ฟังก์ชันโหลดสคริปต์หลัก
local function loadScript()
    statusLabel.Text = กำลังพยายามเชื่อมต่อ...
    progressLabel.Text = โปรดรอ...
    
    local success = false
    local encoded = 
    
    -- ลองทีละลิงก์สำรอง
    for i, url in ipairs(backupURLs) do
        statusLabel.Text = ลองเชื่อมต่อเซิร์ฟเวอร์ที่  .. i
        progressLabel.Text = URL  .. string.sub(url, 1, 30) .. ...
        
        success, encoded = tryDownload(url)
        if success then
            break
        end
        wait(1) -- รอระหว่างการลองเชื่อมต่อ
    end

    if success then
        statusLabel.Text = เชื่อมต่อสำเร็จ!
        progressLabel.Text = กำลังประมวลผล...
        
        -- ถอดรหัสและรันสคริปต์
        local decodeSuccess, decodedScript = pcall(base64Decode, encoded)
        if decodeSuccess then
            local runSuccess, runErr = pcall(function()
                loadstring(decodedScript)()
            end)
            
            if runSuccess then
                statusLabel.Text = โหลดสคริปต์สำเร็จ!
                progressLabel.Text = กำลังปิดหน้าต่าง...
                wait(1.5)
                screenGuiDestroy()
            else
                statusLabel.Text = ข้อผิดพลาดในการรันสคริปต์
                progressLabel.Text = runErr
                retryButton.Visible = true
            end
        else
            statusLabel.Text = ข้อผิดพลาดในการถอดรหัส
            progressLabel.Text = รูปแบบไฟล์ไม่ถูกต้อง
            retryButton.Visible = true
        end
    else
        statusLabel.Text = ไม่สามารถเชื่อมต่อได้
        progressLabel.Text = ตรวจสอบการเชื่อมต่ออินเทอร์เน็ต
        retryButton.Visible = true
    end
end

retryButton.MouseButton1ClickConnect(function()
    retryButton.Visible = false
    loadScript()
end)

-- เริ่มโหลดสคริปต์
